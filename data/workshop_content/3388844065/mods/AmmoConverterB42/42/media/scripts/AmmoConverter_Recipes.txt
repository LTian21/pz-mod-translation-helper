/*
 * Ammo Converter Recipes
 * 
 * This file contains recipes for converting ammunition types.
 * It is divided into two sections:
 * 1. Loose Ammo Conversion: Converts any single bullet to another type.
 * 2. Boxed Ammo Conversion: Converts any full ammo box to another type.
 */
module Base

    /* ========================================================================= */
    /* --- LOOSE AMMO CONVERSION                                             --- */
    /* ========================================================================= */
    /* Converts any item from the defined list into a specific loose bullet type. */
    
    craftRecipe ConvertTo9mm
    {
        timedAction = OpenAmmoBox,
        Time        = 20,
        category    = Ammunition,
        Tags        = InHandCraft;CanBeDoneInDark,
        
        inputs
        {
            item 1 [Base.Bullets45;Base.Bullets44;Base.Bullets38;Base.Bullets357;Base.3030Bullets;Base.308Bullets;Base.556Bullets;Base.ShotgunShells],
        }
        outputs
        {
            item 1 Base.Bullets9mm,
        }
    }
